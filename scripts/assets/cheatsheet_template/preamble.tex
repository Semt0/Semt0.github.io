% --- Preamble for Cheatsheet ---
\usepackage[a4paper, landscape, margin=0.5cm]{geometry} % A4 横向，窄边距
\usepackage{amsmath}          % 数学公式增强
\usepackage{amssymb}          % 数学符号 (如 \mathbb)
\usepackage{graphicx}         % 支持插入图片
\usepackage{multicol}         % 支持多栏排版
\setlength{\columnsep}{0.8em}  % 栏间距
\usepackage{parskip}          % 使用段落间距代替首行缩进
\setlength{\parskip}{0.25em plus 0.05em minus 0.05em} % 小段间距
\setlength{\parindent}{0pt}
\linespread{1.05}              % 舒适行距
\usepackage{ctex}             % ctex 中文支持
\usepackage[svgnames]{xcolor} % 颜色支持
\graphicspath{{source/}}

% 章节标题：section 保留编号、蓝色、紧凑；subsection 不编号
\setcounter{secnumdepth}{1}
\usepackage{titlesec}
\titlespacing*{\section}{0pt}{0.8em plus 0.1em minus 0.1em}{0.3em plus 0.05em minus 0.05em}
\titlespacing*{\subsection}{0pt}{0.5em plus 0.1em minus 0.1em}{0.15em plus 0.05em minus 0.05em}
\titleformat{\section}{\normalfont\bfseries\normalsize\color{NavyBlue}}{\thesection}{0.4em}{}
\titleformat{\subsection}{\normalfont\bfseries\small\color{RoyalBlue}}{\thesubsection}{0.4em}{}
